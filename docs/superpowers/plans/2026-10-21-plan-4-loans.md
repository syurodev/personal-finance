```ts
import { Decimal } from 'decimal.js';
import { addMonths, daysInMonth, monthOf } from '../common/periods.js';
import { type Currency, currencyScale } from '../money/currency.js';

/** Chia `total` thành `count` kỳ bằng nhau (làm tròn xuống theo currency); kỳ cuối nhận phần dư. */
export function splitEvenly(total: Decimal.Value, count: number, currency: Currency): Decimal[] {
  const amount = new Decimal(total);
  const each = amount.div(count).toDecimalPlaces(currencyScale(currency), Decimal.ROUND_DOWN);
  return Array.from({ length: count }, (_, index) => (index === count - 1 ? amount.minus(each.times(count - 1)) : each));
}

/** Ngày đến hạn của `count` kỳ: cùng ngày trong tháng với kỳ đầu, kẹp về ngày cuối tháng. */
export function dueDates(firstDue: string, count: number): string[] {
  const day = Number(firstDue.slice(8, 10));
  return Array.from({ length: count }, (_, index) => {
    const month = addMonths(monthOf(firstDue), index);
    return `${month}-${String(Math.min(day, daysInMonth(month))).padStart(2, '0')}`;
  });
}

export function addOneMonth(date: string): string {
  return dueDates(date, 2)[1]!;
}
```

```ts
/** Tìm danh mục cấp 1 cùng kind theo tên (không dấu); chưa có thì tạo. Dùng cho danh mục hệ thống. */
export async function ensureCategory(name: string, kind: CategoryKind): Promise<CategoryRow> {
  const key = normalizeName(name);
  const existing = (await loadCategories()).find(
    (row) => row.kind === kind && row.parentId === null && normalizeName(row.name) === key,
  );
  if (existing) return existing;
  const created = await createCategory({ name, kind });
  return (await loadCategories()).find((row) => row.id === created.id)!;
}
```

```ts
export function toInstallmentView(row: InstallmentRow, currency: Currency, today: string): InstallmentView {
  const paid = row.status === 'paid';
  return {
    seq: row.seq,
    due_date: row.dueDate,
    amount: formatAmount(paid ? (row.amountPaid ?? row.amountDue) : row.amountDue, currency),
    status: paid ? 'paid' : row.dueDate < today ? 'overdue' : 'due',
    paid_on: row.paidOn,
  };
}

export function remainingOf(rows: readonly InstallmentRow[]): Decimal {
  return rows.filter((row) => row.status === 'due').reduce((sum, row) => sum.plus(row.amountDue), new Decimal(0));
}

function summarize(row: LoanRow, rows: InstallmentRow[], accounts: Map<string, AccountRow>, today: string): LoanSummary {
  const currency = parseCurrency(row.currency);
  const due = rows.filter((r) => r.status === 'due');
  const paid = rows.filter((r) => r.status === 'paid');
  const ref = (id: string) => ({ id, name: accounts.get(id)?.name ?? '' });
  return {
    id: row.id,
    name: row.name,
    direction: row.direction,
    status: row.status,
    currency,
    total_payable: formatAmount(row.totalPayable, currency),
    term_months: row.termMonths,
    start_date: row.startDate,
    first_due_date: row.firstDueDate,
    account: ref(row.accountId),
    payment_account: row.paymentAccountId ? ref(row.paymentAccountId) : null,
    remaining_amount: formatAmount(remainingOf(rows), currency),
    paid_amount: formatAmount(
      paid.reduce((sum, r) => sum.plus(r.amountPaid ?? r.amountDue), new Decimal(0)),
      currency,
    ),
    paid_periods: paid.length,
    remaining_periods: due.length,
    next_due: due[0] ? toInstallmentView(due[0], currency, today) : null,
  };
}

export type CreateLoanInput = {
  name: string;
  direction: LoanDirection;
  /** Một trong hai: tiền mỗi kỳ, hoặc tổng phải trả (chia đều). */

  totalPayable?: string;

  startDate?: string;
  firstDueDate?: string;

  disbursement: 'cash' | 'purchase' | 'existing';
  /** cash: ví nhận tiền vay (vay) hoặc ví chi tiền (cho vay). */
  disbursementAccount?: string;
  /** cash: tiền thực nhận/thực đưa; mặc định = tổng phải trả. Phần chênh lệch là chi phí vay. */
  cashAmount?: string;
  /** purchase: danh mục chi của món hàng. */
  purchaseCategory?: string;
  /** existing: số kỳ đã trả trước khi theo dõi. */
  paidPeriods?: number;
  /** Ví trả nợ (vay) / ví nhận tiền trả (cho vay) mặc định; cash mặc định là disbursementAccount. */
  paymentAccount?: string;
  currency?: string;
};

export async function createLoan(input: CreateLoanInput): Promise<LoanDetail> {
  const { tx, userId } = currentContext();
  const settings = await getUserSettings();
  const name = requireName(input.name);
  if ((await loadLoans()).some((row) => normalizeName(row.name) === normalizeName(name))) {
    throw new DomainError('DUPLICATE_NAME', `Đã có khoản vay "${name}".`);
  }
  const termMonths = input.termMonths;
  if (!Number.isInteger(termMonths) || termMonths < 1 || termMonths > 600) throw invalid('term_months phải là số nguyên từ 1 đến 600.');
  if ((input.installmentAmount === undefined) === (input.totalPayable === undefined)) {
    throw invalid('Truyền đúng một trong hai: installment_amount (tiền mỗi kỳ) hoặc total_payable (tổng phải trả).');
  }
  if (input.disbursement === 'purchase' && input.direction === 'lend') throw invalid('Cho vay không dùng disbursement=purchase.');
  if (input.disbursement !== 'existing' && input.paidPeriods) throw invalid('paid_periods chỉ dùng với disbursement=existing.');
  if (input.disbursement !== 'cash' && input.cashAmount !== undefined) throw invalid('cash_amount chỉ dùng với disbursement=cash.');
  if (input.disbursement === 'cash' && !input.disbursementAccount) throw invalid('Cần disbursement_account khi disbursement=cash.');
  if (input.disbursement === 'purchase' && !input.purchaseCategory) throw invalid('Cần purchase_category khi disbursement=purchase.');

  const startDate = input.startDate ? parseIsoDate(input.startDate, 'start_date') : todayIn(settings.timezone);
  const firstDueDate = input.firstDueDate ? parseIsoDate(input.firstDueDate, 'first_due_date') : addOneMonth(startDate);
  if (firstDueDate <= startDate) throw invalid('first_due_date phải sau start_date.');

  const disbursementAccount = input.disbursement === 'cash' ? await resolveAccount(input.disbursementAccount!) : null;
  const purchaseCategory = input.disbursement === 'purchase' ? await resolveCategory(input.purchaseCategory!, 'expense') : null;
  const paymentAccount = input.paymentAccount ? await resolveAccount(input.paymentAccount) : disbursementAccount;

  const currency = parseCurrency(input.currency ?? disbursementAccount?.currency ?? paymentAccount?.currency ?? settings.baseCurrency);
  for (const account of [disbursementAccount, paymentAccount]) {
    if (account && account.currency !== currency) {
      throw new DomainError('CURRENCY_MISMATCH', `Ví "${account.name}" dùng ${account.currency}, khoản vay dùng ${currency}.`);
    }
  }

  const amounts =
    input.installmentAmount !== undefined
      ? Array.from({ length: termMonths }, () => parseAmount(input.installmentAmount!, currency))
      : splitEvenly(parseAmount(input.totalPayable!, currency), termMonths, currency);
  const total = amounts.reduce((sum, a) => sum.plus(a), new Decimal(0));
  const paidPeriods = input.paidPeriods ?? 0;
  if (!Number.isInteger(paidPeriods) || paidPeriods < 0 || paidPeriods >= termMonths) {
    throw invalid('paid_periods phải từ 0 đến term_months − 1.');
  }
  const cash = disbursementAccount ? (input.cashAmount !== undefined ? parseAmount(input.cashAmount, currency) : total) : null;
  if (cash && cash.greaterThan(total)) throw invalid('cash_amount không được lớn hơn tổng phải trả.');

  const remaining = amounts.slice(paidPeriods).reduce((sum, a) => sum.plus(a), new Decimal(0));
  const sign = input.direction === 'borrow' ? -1 : 1;
  const account = await createAccount({
    type: 'loan',
    name,
    currency,
    openingBalance: input.disbursement === 'existing' ? remaining.times(sign).toFixed() : '0',
  });

  let disbursementTransactionId: string | null = null;
  let costTransactionId: string | null = null;
  if (disbursementAccount && cash) {
    const transfer =
      input.direction === 'borrow'
        ? await recordTransfer({ fromAccount: account.id, toAccount: disbursementAccount.id, amount: cash.toFixed(), occurredOn: startDate, note: `Giải ngân ${name}` })
        : await recordTransfer({ fromAccount: disbursementAccount.id, toAccount: account.id, amount: cash.toFixed(), occurredOn: startDate, note: `Cho vay ${name}` });
    disbursementTransactionId = transfer.from.id;
    const cost = total.minus(cash);
    if (cost.greaterThan(0)) {
      costTransactionId = await recordLoanCost({ direction: input.direction, accountId: account.id }, cost, startDate, `Lãi & phí ${name}`);
    }
  } else if (purchaseCategory) {
    const { created } = await recordTransactions([
      { amount: total.toFixed(), type: 'expense', account: account.id, category: purchaseCategory.id, occurredOn: startDate, description: name },
    ]);
    disbursementTransactionId = created[0]!.id;
  }

  const dates = dueDates(firstDueDate, termMonths);
  const [row] = await tx
    .insert(loan)
    .values({
      userId,
      name,
      direction: input.direction,
      totalPayable: total.toFixed(),
      currency,
      termMonths,
      startDate,
      firstDueDate,
      accountId: account.id,
      paymentAccountId: paymentAccount?.id ?? null,
      disbursementTransactionId,
      costTransactionId,
    })
    .returning();
  await tx.insert(loanInstallment).values(
    amounts.map((amount, index) => ({
      userId,
      loanId: row!.id,
      seq: index + 1,
      dueDate: dates[index]!,
      amountDue: amount.toFixed(),
      status: index < paidPeriods ? ('paid' as const) : ('due' as const),
    })),
  );
  await recordAudit('loan.create', 'loan', [row!.id]);
  return loanDetail(row!.id);
}

export async function getLoanStatus(input: { loan?: string; includeInactive?: boolean }): Promise<{
  loans: LoanSummary[];
  schedule: InstallmentView[] | null;
}> {

export async function listUpcomingPayments(input: { asOf?: string; days?: number } = {}): Promise<UpcomingPayments> {
  const { tx } = currentContext();
  const asOf = input.asOf ? parseIsoDate(input.asOf, 'as_of') : await today();
  const days = input.days ?? 30;
  if (!Number.isInteger(days) || days < 1 || days > 366) throw invalid('days phải từ 1 đến 366.');
  const until = addDays(asOf, days);
  const loans = await loadLoans();
  if (loans.length === 0) return { as_of: asOf, until, items: [], totals: [], overdue_count: 0 };
  const order = new Map(loans.map((row, index) => [row.id, index]));
  const byId = new Map(loans.map((row) => [row.id, row]));

  const rows = await tx
    .select()
    .from(loanInstallment)
    .where(and(eq(loanInstallment.status, 'due'), lte(loanInstallment.dueDate, until), inArray(loanInstallment.loanId, [...byId.keys()])));
  rows.sort((a, b) => a.dueDate.localeCompare(b.dueDate) || order.get(a.loanId)! - order.get(b.loanId)! || a.seq - b.seq);

  const items = rows.map((row) => {
    const owner = byId.get(row.loanId)!;
    const currency = parseCurrency(owner.currency);
    return { loan: { id: owner.id, name: owner.name }, direction: owner.direction, currency, ...toInstallmentView(row, currency, asOf) };
  });

  const totals = new Map<string, { currency: Currency; direction: LoanDirection; amount: Decimal }>();
  for (const item of items) {
    const key = `${item.currency}|${item.direction}`;
    const entry = totals.get(key) ?? { currency: parseCurrency(item.currency), direction: item.direction, amount: new Decimal(0) };
    entry.amount = entry.amount.plus(item.amount);
    totals.set(key, entry);
  }

  return {
    as_of: asOf,
    until,
    items,
    totals: [...totals.values()]
      .sort((a, b) => a.currency.localeCompare(b.currency) || a.direction.localeCompare(b.direction))
      .map((t) => ({ currency: t.currency, direction: t.direction, amount: formatAmount(t.amount, t.currency) })),
    overdue_count: items.filter((item) => item.status === 'overdue').length,
  };
}

/** Tóm tắt nợ cho get_overview; null khi không có khoản vay đang chạy. */
export async function debtSummary(asOf: string): Promise<DebtSummary | null> {
  const loans = await loadLoans();
  if (loans.length === 0) return null;
  const order = new Map(loans.map((row, index) => [row.id, index]));
  const byId = new Map(loans.map((row) => [row.id, row]));
  const rows = (await installmentsOf(loans.map((l) => l.id))).filter((r) => r.status === 'due');

  const perCurrency = new Map<Currency, { borrowed: Decimal; lent: Decimal }>();
  for (const row of rows) {
    const owner = byId.get(row.loanId)!;
    const currency = parseCurrency(owner.currency);
    const entry = perCurrency.get(currency) ?? { borrowed: new Decimal(0), lent: new Decimal(0) };
    if (owner.direction === 'borrow') entry.borrowed = entry.borrowed.plus(row.amountDue);
    else entry.lent = entry.lent.plus(row.amountDue);
    perCurrency.set(currency, entry);
  }

  const next = [...rows].sort(
    (a, b) => a.dueDate.localeCompare(b.dueDate) || order.get(a.loanId)! - order.get(b.loanId)! || a.seq - b.seq,
  )[0];
  const nextOwner = next ? byId.get(next.loanId)! : null;
  return {
    items: [...perCurrency.entries()]
      .sort(([a], [b]) => a.localeCompare(b))
      .map(([currency, entry]) => ({ currency, borrowed: formatAmount(entry.borrowed, currency), lent: formatAmount(entry.lent, currency) })),
    next_due:
      next && nextOwner
        ? {
            loan: nextOwner.name,
            direction: nextOwner.direction,
            due_date: next.dueDate,
            amount: formatAmount(next.amountDue, parseCurrency(nextOwner.currency)),
            currency: nextOwner.currency,
          }
        : null,
    overdue_count: rows.filter((row) => row.dueDate < asOf).length,
  };
}
```

```ts
/** Khoá dòng khoản vay tới hết transaction: thao tác song song phải chờ và đọc lịch đã cập nhật. */
async function lockLoan(id: string): Promise<void> {
  await currentContext().tx.execute(sql`select 1 from loan where id = ${id} for update`);
}

async function activeLoan(ref: string): Promise<LoanRow> {
  const target = await resolveLoan(ref, { includeInactive: true });
  if (target.status !== 'active') {
    throw new DomainError('LOAN_CLOSED', `Khoản vay "${target.name}" đã ${target.status === 'closed' ? 'tất toán' : 'bị huỷ'}.`);
  }
  await lockLoan(target.id);
  return target;
}

/** Dòng điều chỉnh trên ví nợ (không tính vào thu/chi), giữ bất biến số dư = các kỳ chưa trả. */
async function adjustDebt(target: LoanRow, signedAmount: Decimal, occurredOn: string, note: string): Promise<void> {
  const { tx, userId, clientId } = currentContext();
  const [row] = await tx
    .insert(ledgerTransaction)
    .values({
      userId,
      accountId: target.accountId,
      kind: 'adjustment',
      amount: signedAmount.toFixed(),
      currency: target.currency,
      occurredOn,
      note,
      sourceClientId: clientId,
    })
    .returning();
  await recordAudit('loan.adjustment', 'ledger_transaction', [row!.id]);
}

async function today(): Promise<string> {

export type LoanPaymentInput = {
  loan: string;
  paidOn?: string;
  /** Ví trả (vay) / ví nhận (cho vay); mặc định ví đã đặt cho khoản vay. */
  account?: string;
  /** Số tiền thực trả; mặc định = tiền kỳ (hoặc toàn bộ phần còn lại khi payoff). */
  amount?: string;
  /** Tất toán: trả `amount`, phần còn lại được miễn và khoản vay đóng lại. */
  payoff?: boolean;
};

export async function recordLoanPayment(input: LoanPaymentInput): Promise<{
  loan: LoanSummary;
  paid: InstallmentView;
  balances: AccountBalanceView[];
}> {
  const { tx } = currentContext();
  const target = await activeLoan(input.loan);
  const currency: Currency = parseCurrency(target.currency);
  const borrow = target.direction === 'borrow';
  const due = (await installmentsOf([target.id])).filter((row) => row.status === 'due');
  const next = due[0];
  if (!next) throw new DomainError('LOAN_CLOSED', `Khoản vay "${target.name}" không còn kỳ nào phải trả.`);

  const account = input.account ? await resolveAccount(input.account) : target.paymentAccountId ? await accountById(target.paymentAccountId) : null;
  if (!account) throw new DomainError('VALIDATION_ERROR', `Khoản vay "${target.name}" chưa có ví trả mặc định; truyền account.`);
  if (account.currency !== target.currency) {
    throw new DomainError('CURRENCY_MISMATCH', `Ví "${account.name}" dùng ${account.currency}, khoản vay dùng ${target.currency}.`);
  }

  const remaining = remainingOf(due);
  const amount =
    input.amount !== undefined ? parseAmount(input.amount, currency) : input.payoff ? remaining : new Decimal(next.amountDue);
  if (amount.greaterThan(remaining)) {
    throw new DomainError(
      'OVERPAYMENT',
      `Số tiền ${formatAmount(amount, currency)} vượt phần còn phải trả ${formatAmount(remaining, currency)}.`,
      { remaining_amount: formatAmount(remaining, currency) },
    );
  }
  const paidOn = input.paidOn ? parseIsoDate(input.paidOn, 'paid_on') : await today();
  const note = `${target.name} kỳ ${next.seq}`;

  const transfer = borrow
    ? await recordTransfer({ fromAccount: account.id, toAccount: target.accountId, amount: amount.toFixed(), occurredOn: paidOn, note: `Trả ${note}` })
    : await recordTransfer({ fromAccount: target.accountId, toAccount: account.id, amount: amount.toFixed(), occurredOn: paidOn, note: `Thu ${note}` });

  const [paidRow] = await tx
    .update(loanInstallment)
    .set({ status: 'paid', paidOn, amountPaid: amount.toFixed(), transferGroupId: transfer.transfer_group_id })
    .where(eq(loanInstallment.id, next.id))
    .returning();

  const rest = due.slice(1);
  const left = remaining.minus(amount);
  if (input.payoff || left.isZero()) {
    if (left.greaterThan(0)) {
      // Phần được miễn khi tất toán sớm.
      await adjustDebt(target, borrow ? left : left.negated(), paidOn, `Miễn giảm khi tất toán ${target.name}`);
    }
    if (rest.length > 0) await tx.delete(loanInstallment).where(inArray(loanInstallment.id, rest.map((r) => r.id)));
    await tx.update(loan).set({ status: 'closed', closedAt: new Date() }).where(eq(loan.id, target.id));
  } else {
    let diff = amount.minus(next.amountDue);
    if (diff.greaterThan(0)) {
      // Trả dư: trừ dần vào các kỳ cuối.
      for (const row of [...rest].reverse()) {
        if (diff.isZero()) break;
        const cut = Decimal.min(diff, row.amountDue);
        const reduced = new Decimal(row.amountDue).minus(cut);
        if (reduced.isZero()) await tx.delete(loanInstallment).where(eq(loanInstallment.id, row.id));
        else await tx.update(loanInstallment).set({ amountDue: reduced.toFixed() }).where(eq(loanInstallment.id, row.id));
        diff = diff.minus(cut);
      }
    } else if (diff.lessThan(0)) {
      // Trả thiếu: cộng vào kỳ kế tiếp, hoặc thêm một kỳ tháng sau.
      const shortfall = diff.negated();
      if (rest[0]) {
        await tx
          .update(loanInstallment)
          .set({ amountDue: new Decimal(rest[0].amountDue).plus(shortfall).toFixed() })
          .where(eq(loanInstallment.id, rest[0].id));
      } else {
        await tx.insert(loanInstallment).values({
          userId: target.userId,
          loanId: target.id,
          seq: next.seq + 1,
          dueDate: addOneMonth(next.dueDate),
          amountDue: shortfall.toFixed(),
        });
      }
    }
  }

  await recordAudit('loan.payment', 'loan_installment', [next.id]);
  return {
    loan: withoutSchedule(await loanDetail(target.id)),
    paid: toInstallmentView(paidRow!, currency, paidOn),
    balances: await balancesFor([account.id, target.accountId]),
  };
}

export async function updateLoan(input: {
  loan: string;
  name?: string;
  /** Số tiền mới cho mọi kỳ chưa trả (ví dụ ngân hàng đổi lãi suất). */
  installmentAmount?: string;
  paymentAccount?: string;
}): Promise<LoanDetail> {
  const { tx } = currentContext();
  if (input.name === undefined && input.installmentAmount === undefined && input.paymentAccount === undefined) {
    throw new DomainError('VALIDATION_ERROR', 'Không có trường nào để sửa.');
  }
  const target = await activeLoan(input.loan);
  const currency = parseCurrency(target.currency);
  const changes: Partial<typeof loan.$inferInsert> = {};

  if (input.name !== undefined) {
    const name = requireName(input.name);
    const clash = (await loadLoans()).find((row) => row.id !== target.id && normalizeName(row.name) === normalizeName(name));
    if (clash) throw new DomainError('DUPLICATE_NAME', `Đã có khoản vay "${clash.name}".`);
    await updateAccount(target.accountId, { name });
    changes.name = name;
  }
  if (input.paymentAccount !== undefined) {
    const account = await resolveAccount(input.paymentAccount);
    if (account.currency !== target.currency) {
      throw new DomainError('CURRENCY_MISMATCH', `Ví "${account.name}" dùng ${account.currency}, khoản vay dùng ${target.currency}.`);
    }
    changes.paymentAccountId = account.id;
  }
  if (input.installmentAmount !== undefined) {
    const amount = parseAmount(input.installmentAmount, currency);
    const due = (await installmentsOf([target.id])).filter((row) => row.status === 'due');
    const before = remainingOf(due);
    await tx.update(loanInstallment).set({ amountDue: amount.toFixed() }).where(inArray(loanInstallment.id, due.map((r) => r.id)));
    const delta = amount.times(due.length).minus(before);
    const date = await today();
    if (delta.greaterThan(0)) {
      await recordLoanCost(target, delta, date, `Tăng tiền kỳ ${target.name}`);
    } else if (delta.lessThan(0)) {
      const reduction = delta.negated();
      await adjustDebt(target, target.direction === 'borrow' ? reduction : reduction.negated(), date, `Giảm tiền kỳ ${target.name}`);
    }
  }

  if (Object.keys(changes).length > 0) await tx.update(loan).set(changes).where(eq(loan.id, target.id));
  await recordAudit('loan.update', 'loan', [target.id]);
  return loanDetail(target.id);
}

export async function cancelLoan(ref: string): Promise<{ cancelled: true; loan: LoanSummary }> {
  const { tx } = currentContext();
  const target = await activeLoan(ref);
  if ((await installmentsOf([target.id])).some((row) => row.transferGroupId)) {
    throw new DomainError(
      'LOAN_HAS_PAYMENTS',
      `Khoản vay "${target.name}" đã có kỳ trả được ghi nên không huỷ được. Nếu cần, xoá từng giao dịch bằng delete_transaction.`,
    );
  }
  for (const id of [target.disbursementTransactionId, target.costTransactionId]) {
    if (id) await deleteTransaction(id);
  }
  await updateAccount(target.accountId, { archived: true });
  await tx.update(loan).set({ status: 'cancelled', closedAt: new Date() }).where(eq(loan.id, target.id));
  await recordAudit('loan.cancel', 'loan', [target.id]);
  return { cancelled: true, loan: withoutSchedule(await loanDetail(target.id)) };
}
```

```ts
import type { Client } from '@modelcontextprotocol/sdk/client/index.js';
import type { NestExpressApplication } from '@nestjs/platform-express';
import { auth } from '../src/auth/auth.js';
import { startApp } from './app.js';
import { ALL_TOOLS, callError, callOk, connectMcp } from './mcp-client.js';
import { authorizeAndExchange, registerClient } from './oauth-flow.js';

const credentials = { email: 'loans-e2e@example.com', password: 'password-123' };

describe('loans over MCP', () => {
  let app: NestExpressApplication;
  let baseUrl: string;
  let client: Client;

  const balance = async (name: string): Promise<string> =>
    (await callOk(client, 'get_overview')).accounts.find((a: { name: string }) => a.name === name).balance;

  beforeAll(async () => {
    ({ app, baseUrl } = await startApp());
    await auth.api.signUpEmail({ body: { name: 'Loans E2E', ...credentials } });
    const clientId = await registerClient(baseUrl);
    const tokens = await authorizeAndExchange(baseUrl, clientId, credentials);
    client = await connectMcp(baseUrl, tokens.access_token);
    const { tools } = await client.listTools();
    expect(tools.map((t) => t.name).sort()).toEqual([...ALL_TOOLS].sort());
    await callOk(client, 'create_account', { type: 'bank', name: 'VCB', opening_balance: '10000000' });
  });

  afterAll(async () => {
    await client?.close();
    await app.close();
  });

  it('creates a car loan, pays an installment, changes the amount and lists what is due', async () => {
    const created = await callOk(client, 'create_loan', {
      name: 'Vay mua xe',
      direction: 'borrow',
      installment_amount: '5500000',
      term_months: 6,
      start_date: '2026-07-15',
      disbursement: 'cash',
      disbursement_account: 'VCB',
      cash_amount: '30000000',
    });
    expect(created).toMatchObject({
      total_payable: '33000000',
      remaining_amount: '33000000',
      account: expect.objectContaining({ name: 'Vay mua xe' }),
      next_due: expect.objectContaining({ seq: 1, due_date: '2026-08-15', amount: '5500000' }),
    });
    expect(created.schedule).toHaveLength(6);

    const overview = await callOk(client, 'get_overview');
    expect(overview.accounts.find((a: { name: string }) => a.name === 'VCB').balance).toBe('40000000');
    expect(overview.debts.items).toEqual([{ currency: 'VND', borrowed: '33000000', lent: '0' }]);

    const paid = await callOk(client, 'record_loan_payment', { loan: 'vay mua xe', paid_on: '2026-08-15' });
    expect(paid.paid).toMatchObject({ seq: 1, status: 'paid', amount: '5500000' });
    expect(paid.loan.remaining_amount).toBe('27500000');
    expect(await balance('VCB')).toBe('34500000');

    // 5 kỳ còn lại × 6.000.000.
    const updated = await callOk(client, 'update_loan', { loan: 'Vay mua xe', installment_amount: '6000000' });
    expect(updated).toMatchObject({ remaining_amount: '30000000', next_due: expect.objectContaining({ seq: 2, amount: '6000000' }) });

    const upcoming = await callOk(client, 'list_upcoming_payments', { as_of: '2026-09-20', days: 30 });
    expect(upcoming.items.map((i: any) => [i.seq, i.due_date, i.status, i.amount])).toEqual([
      [2, '2026-09-15', 'overdue', '6000000'],
      [3, '2026-10-15', 'due', '6000000'],
    ]);
    expect(upcoming.totals).toEqual([{ currency: 'VND', direction: 'borrow', amount: '12000000' }]);

    const status = await callOk(client, 'get_loan_status', { loan: 'Vay mua xe' });
    expect(status.loans[0]).toMatchObject({ paid_periods: 1, remaining_periods: 5 });
    expect(status.schedule[0].status).toBe('paid');
  });

  it('cancels a mistaken loan and protects loans with payments', async () => {
    await callOk(client, 'create_loan', {
      name: 'Vay nhầm',
      direction: 'borrow',
      total_payable: '1000000',
      term_months: 2,
      disbursement: 'cash',
      disbursement_account: 'VCB',
    });
    expect(await balance('VCB')).toBe('35500000');
    await callOk(client, 'cancel_loan', { loan: 'Vay nhầm' });
    expect(await balance('VCB')).toBe('34500000');
    expect((await callOk(client, 'get_loan_status')).loans.map((l: any) => l.name)).toEqual(['Vay mua xe']);

    expect((await callError(client, 'cancel_loan', { loan: 'Vay mua xe' })).code).toBe('LOAN_HAS_PAYMENTS');
    const overpay = await callError(client, 'record_loan_payment', { loan: 'Vay mua xe', amount: '999999999' });
    expect(overpay.code).toBe('OVERPAYMENT');
    expect(overpay.details.remaining_amount).toBe('30000000');
  });

  it('a read-only connection cannot record payments', async () => {
    const clientId = await registerClient(baseUrl);
    const tokens = await authorizeAndExchange(baseUrl, clientId, credentials, 'offline_access finance:read');
    const readOnly = await connectMcp(baseUrl, tokens.access_token);
    try {
      await readOnly.listTools();
      expect((await callError(readOnly, 'record_loan_payment', { loan: 'Vay mua xe' })).code).toBe('INSUFFICIENT_SCOPE');
      expect((await callOk(readOnly, 'get_loan_status')).loans[0].paid_periods).toBe(1);
    } finally {
      await readOnly.close();
    }
  });
});
```
