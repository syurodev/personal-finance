```ts
import { Decimal } from 'decimal.js';
import { addMonths, daysInMonth, monthOf } from '../common/periods.js';
import { type Currency, currencyScale } from '../money/currency.js';

/** Tạm thời; Task 2 chuyển sang `import type { LoanMethod } from '../db/schema/index.js'`. */
type LoanMethod = 'annuity' | 'equal_principal' | 'flat';

/** Đủ chữ số cho (1 + r)^-n; tiền làm tròn half-up theo số lẻ của currency. */
const Precise = Decimal.clone({ precision: 40 });

export type InstallmentAmount = { principal: Decimal; interest: Decimal };

/**
 * Chia khoản vay thành `count` kỳ tháng.
 * - equal_principal: gốc đều, lãi = dư nợ × r
 * - annuity: tổng gốc + lãi bằng nhau mỗi kỳ, lãi = dư nợ × r
 * - flat: gốc đều, lãi = gốc ban đầu × r
 * r = annualRate / 1200. Kỳ cuối nhận phần gốc còn lại để tổng gốc đúng bằng khoản vay.
 */
export function computeInstallments(input: {
  principal: Decimal.Value;
  annualRate: Decimal.Value;
  count: number;
  method: LoanMethod;
  currency: Currency;
}): InstallmentAmount[] {
  const scale = currencyScale(input.currency);
  const round = (value: Decimal) => value.toDecimalPlaces(scale, Decimal.ROUND_HALF_UP);
  const principal = new Precise(input.principal);
  const rate = new Precise(input.annualRate).div(1200);
  const count = input.count;
  const evenPrincipal = round(principal.div(count));
  const payment =
    input.method !== 'annuity'
      ? null
      : round(rate.isZero() ? principal.div(count) : principal.times(rate).div(new Precise(1).minus(rate.plus(1).pow(-count))));

  const rows: InstallmentAmount[] = [];
  let balance = principal;
  for (let period = 1; period <= count; period++) {
    const interest = round(input.method === 'flat' ? principal.times(rate) : balance.times(rate));
    const part = period === count ? balance : payment ? payment.minus(interest) : evenPrincipal;
    rows.push({ principal: new Decimal(part), interest: new Decimal(interest) });
    balance = balance.minus(part);
  }
  return rows;
}

/** Ngày đến hạn của `count` kỳ: cùng ngày trong tháng với kỳ đầu, kẹp về ngày cuối tháng. */
export function dueDates(firstDue: string, count: number): string[] {
  const day = Number(firstDue.slice(8, 10));
  return Array.from({ length: count }, (_, index) => {
    const month = addMonths(monthOf(firstDue), index);
    return `${month}-${String(Math.min(day, daysInMonth(month))).padStart(2, '0')}`;
  });
}

export function addOneMonth(date: string): string {
  return dueDates(date, 2)[1]!;
}
```

```ts
/** Tìm danh mục cấp 1 cùng kind theo tên (không dấu); chưa có thì tạo. Dùng cho danh mục hệ thống. */
export async function ensureCategory(name: string, kind: CategoryKind): Promise<CategoryRow> {
  const key = normalizeName(name);
  const existing = (await loadCategories()).find(
    (row) => row.kind === kind && row.parentId === null && normalizeName(row.name) === key,
  );
  if (existing) return existing;
  const created = await createCategory({ name, kind });
  return (await loadCategories()).find((row) => row.id === created.id)!;
}
```

```ts
export function parseRate(input: string): Decimal {
  const rate = parsePositiveDecimal(input, 'annual_rate');
  if (rate.greaterThanOrEqualTo(100)) throw new DomainError('VALIDATION_ERROR', 'annual_rate là % năm, phải nhỏ hơn 100.');
  return rate;
}

function parseTerm(value: number): number {
  if (!Number.isInteger(value) || value < 1 || value > 600) {
    throw new DomainError('VALIDATION_ERROR', 'term_months phải là số nguyên từ 1 đến 600.');
  }
  return value;
}

function required(value: string | undefined, field: string, reason: string): string {
  if (!value) throw new DomainError('VALIDATION_ERROR', `Cần ${field} khi ${reason}.`);
  return value;

export async function loadLoans(options: { includeInactive?: boolean } = {}): Promise<LoanRow[]> {
  const { tx } = currentContext();

/** Tìm khoản vay đang chạy theo id/tên; includeInactive để tìm cả khoản đã tất toán/huỷ. */
export async function resolveLoan(ref: string, options: { includeInactive?: boolean } = {}): Promise<LoanRow> {
  try {
    return resolveByName(await loadLoans(), ref, 'LOAN');
  } catch (error) {

export function toInstallmentView(row: InstallmentRow, currency: Currency, today: string): InstallmentView {
  const paid = row.status === 'paid';
  const principal = new Decimal(paid ? (row.principalPaid ?? row.principalDue) : row.principalDue);
  const interest = new Decimal(paid ? (row.interestPaid ?? row.interestDue) : row.interestDue);
  return {
    seq: row.seq,
    due_date: row.dueDate,
    principal: formatAmount(principal, currency),
    interest: formatAmount(interest, currency),
    total: formatAmount(principal.plus(interest), currency),
    status: paid ? 'paid' : row.dueDate < today ? 'overdue' : 'due',
    paid_on: row.paidOn,
  };
}

function sumOf(rows: readonly InstallmentRow[], pick: (row: InstallmentRow) => string): Decimal {
  return rows.reduce((sum, row) => sum.plus(pick(row)), new Decimal(0));
}

export function outstandingOf(rows: readonly InstallmentRow[]): Decimal {
  return sumOf(
    rows.filter((row) => row.status === 'due'),
    (row) => row.principalDue,
  );
}

function summarize(row: LoanRow, rows: InstallmentRow[], accounts: Map<string, AccountRow>, today: string): LoanSummary {
  const currency = parseCurrency(row.currency);
  const due = rows.filter((r) => r.status === 'due');
  const paid = rows.filter((r) => r.status === 'paid');
  const ref = (id: string) => ({ id, name: accounts.get(id)?.name ?? '' });
  return {
    id: row.id,
    name: row.name,
    direction: row.direction,
    method: row.method,
    status: row.status,
    currency,
    principal: formatAmount(row.principal, currency),
    annual_rate: new Decimal(row.annualRate).toFixed(),
    term_months: row.termMonths,
    start_date: row.startDate,
    first_due_date: row.firstDueDate,
    account: ref(row.accountId),
    payment_account: row.paymentAccountId ? ref(row.paymentAccountId) : null,
    outstanding_principal: formatAmount(outstandingOf(rows), currency),
    paid_periods: paid.length,
    remaining_periods: due.length,
    interest_paid: formatAmount(
      sumOf(paid, (r) => r.interestPaid ?? r.interestDue),
      currency,
    ),
    interest_remaining: formatAmount(
      sumOf(due, (r) => r.interestDue),
      currency,
    ),
    next_due: due[0] ? toInstallmentView(due[0], currency, today) : null,
  };
}

async function today(): Promise<string> {
  return todayIn((await getUserSettings()).timezone);
}

export async function loanDetail(loanId: string): Promise<LoanDetail> {
  const row = (await loadLoans({ includeInactive: true })).find((l) => l.id === loanId)!;
  const rows = await installmentsOf([row.id]);

export type CreateLoanInput = {
  name: string;
  direction: LoanDirection;
  method: LoanMethod;
  principal: string;
  annualRate: string;
  termMonths: number;
  startDate?: string;
  firstDueDate?: string;
  /** cash: tiền chuyển qua ví; purchase: mua hàng trả góp; existing: khoản vay đã chạy từ trước. */
  disbursement: 'cash' | 'purchase' | 'existing';
  /** cash: ví nhận tiền vay (vay) hoặc ví chi tiền (cho vay). */
  disbursementAccount?: string;
  /** purchase: danh mục chi của món hàng. */
  purchaseCategory?: string;
  /** existing: số kỳ đã trả trước khi theo dõi. */
  paidPeriods?: number;
  /** Ví trả nợ (vay) / ví nhận tiền trả (cho vay) mặc định; cash mặc định là disbursementAccount. */
  paymentAccount?: string;
  currency?: string;
};

export async function createLoan(input: CreateLoanInput): Promise<LoanDetail> {
  const { tx, userId } = currentContext();
  const settings = await getUserSettings();
  const name = requireName(input.name);
  if ((await loadLoans()).some((row) => normalizeName(row.name) === normalizeName(name))) {
    throw new DomainError('DUPLICATE_NAME', `Đã có khoản vay "${name}".`);
  }
  const termMonths = parseTerm(input.termMonths);
  const annualRate = parseRate(input.annualRate);
  const startDate = input.startDate ? parseIsoDate(input.startDate, 'start_date') : todayIn(settings.timezone);
  const firstDueDate = input.firstDueDate ? parseIsoDate(input.firstDueDate, 'first_due_date') : addOneMonth(startDate);
  if (firstDueDate <= startDate) throw new DomainError('VALIDATION_ERROR', 'first_due_date phải sau start_date.');

  if (input.disbursement === 'purchase' && input.direction === 'lend') {
    throw new DomainError('VALIDATION_ERROR', 'Cho vay không dùng disbursement=purchase; dùng cash hoặc existing.');
  }
  if (input.disbursement !== 'existing' && input.paidPeriods) {
    throw new DomainError('VALIDATION_ERROR', 'paid_periods chỉ dùng với disbursement=existing.');
  }
  const disbursementAccount =
    input.disbursement === 'cash'
      ? await resolveAccount(required(input.disbursementAccount, 'disbursement_account', 'disbursement=cash'))
      : null;
  const purchaseCategory =
    input.disbursement === 'purchase'
      ? await resolveCategory(required(input.purchaseCategory, 'purchase_category', 'disbursement=purchase'), 'expense')
      : null;
  const paymentAccount = input.paymentAccount ? await resolveAccount(input.paymentAccount) : disbursementAccount;

  const currency = parseCurrency(input.currency ?? disbursementAccount?.currency ?? paymentAccount?.currency ?? settings.baseCurrency);
  for (const account of [disbursementAccount, paymentAccount]) {
    if (account && account.currency !== currency) {
      throw new DomainError('CURRENCY_MISMATCH', `Ví "${account.name}" dùng ${account.currency}, khoản vay dùng ${currency}.`);
    }
  }
  const principal = parseAmount(input.principal, currency);
  const paidPeriods = input.paidPeriods ?? 0;
  if (!Number.isInteger(paidPeriods) || paidPeriods < 0 || paidPeriods >= termMonths) {
    throw new DomainError('VALIDATION_ERROR', 'paid_periods phải từ 0 đến term_months − 1.');
  }

  const amounts = computeInstallments({ principal, annualRate, count: termMonths, method: input.method, currency });
  const dates = dueDates(firstDueDate, termMonths);
  const remaining = amounts.slice(paidPeriods).reduce((sum, a) => sum.plus(a.principal), new Decimal(0));
  const sign = input.direction === 'borrow' ? -1 : 1;

  const account = await createAccount({
    type: 'loan',
    name,
    currency,
    openingBalance: input.disbursement === 'existing' ? remaining.times(sign).toFixed() : '0',
  });

  let disbursementTransactionId: string | null = null;
  if (disbursementAccount) {
    const transfer =
      input.direction === 'borrow'
        ? await recordTransfer({ fromAccount: account.id, toAccount: disbursementAccount.id, amount: principal.toFixed(), occurredOn: startDate, note: `Giải ngân ${name}` })
        : await recordTransfer({ fromAccount: disbursementAccount.id, toAccount: account.id, amount: principal.toFixed(), occurredOn: startDate, note: `Cho vay ${name}` });
    disbursementTransactionId = transfer.from.id;
  } else if (purchaseCategory) {
    const { created } = await recordTransactions([
      { amount: principal.toFixed(), type: 'expense', account: account.id, category: purchaseCategory.id, occurredOn: startDate, description: name },
    ]);
    disbursementTransactionId = created[0]!.id;
  }

  const [row] = await tx
    .insert(loan)
    .values({
      userId,
      name,
      direction: input.direction,
      method: input.method,
      principal: principal.toFixed(),
      currency,
      annualRate: annualRate.toFixed(),
      termMonths,
      startDate,
      firstDueDate,
      accountId: account.id,
      paymentAccountId: paymentAccount?.id ?? null,
      disbursementTransactionId,
    })
    .returning();
  await tx.insert(loanInstallment).values(
    amounts.map((amount, index) => ({
      userId,
      loanId: row!.id,
      seq: index + 1,
      dueDate: dates[index]!,
      principalDue: amount.principal.toFixed(),
      interestDue: amount.interest.toFixed(),
      status: index < paidPeriods ? ('paid' as const) : ('due' as const),
    })),
  );
  await recordAudit('loan.create', 'loan', [row!.id]);
  return loanDetail(row!.id);
}

export async function listUpcomingPayments(input: { asOf?: string; days?: number } = {}): Promise<UpcomingPayments> {
  const { tx } = currentContext();
  const asOf = input.asOf ? parseIsoDate(input.asOf, 'as_of') : await today();
  const days = input.days ?? 30;
  if (!Number.isInteger(days) || days < 1 || days > 366) throw new DomainError('VALIDATION_ERROR', 'days phải từ 1 đến 366.');
  const until = addDays(asOf, days);
  const loans = new Map((await loadLoans()).map((row) => [row.id, row]));
  if (loans.size === 0) return { as_of: asOf, until, items: [], totals: [], overdue_count: 0 };

  const rows = await tx
    .select()
    .from(loanInstallment)
    .where(
      and(eq(loanInstallment.status, 'due'), lte(loanInstallment.dueDate, until), inArray(loanInstallment.loanId, [...loans.keys()])),
    )
    .orderBy(asc(loanInstallment.dueDate), asc(loanInstallment.seq));

  const items = rows.map((row) => {
    const owner = loans.get(row.loanId)!;
    const currency = parseCurrency(owner.currency);
    return { loan: { id: owner.id, name: owner.name }, direction: owner.direction, currency, ...toInstallmentView(row, currency, asOf) };
  });

  const totals = new Map<string, { currency: Currency; direction: LoanDirection; principal: Decimal; interest: Decimal }>();
  for (const item of items) {
    const key = `${item.currency}|${item.direction}`;
    const entry = totals.get(key) ?? { currency: parseCurrency(item.currency), direction: item.direction, principal: new Decimal(0), interest: new Decimal(0) };
    entry.principal = entry.principal.plus(item.principal);
    entry.interest = entry.interest.plus(item.interest);
    totals.set(key, entry);
  }

  return {
    as_of: asOf,
    until,
    items,
    totals: [...totals.values()]
      .sort((a, b) => a.currency.localeCompare(b.currency) || a.direction.localeCompare(b.direction))
      .map((t) => ({
        currency: t.currency,
        direction: t.direction,
        principal: formatAmount(t.principal, t.currency),
        interest: formatAmount(t.interest, t.currency),
        total: formatAmount(t.principal.plus(t.interest), t.currency),
      })),
    overdue_count: items.filter((item) => item.status === 'overdue').length,
  };
}

/** Tóm tắt nợ cho get_overview; null khi không có khoản vay đang chạy. */
export async function debtSummary(asOf: string): Promise<DebtSummary | null> {
  const loans = await loadLoans();
  if (loans.length === 0) return null;
  const rows = (await installmentsOf(loans.map((l) => l.id))).filter((r) => r.status === 'due');
  const byId = new Map(loans.map((l) => [l.id, l]));

  const perCurrency = new Map<Currency, { borrowed: Decimal; lent: Decimal }>();
  for (const row of rows) {
    const owner = byId.get(row.loanId)!;
    const currency = parseCurrency(owner.currency);
    const entry = perCurrency.get(currency) ?? { borrowed: new Decimal(0), lent: new Decimal(0) };
    if (owner.direction === 'borrow') entry.borrowed = entry.borrowed.plus(row.principalDue);
    else entry.lent = entry.lent.plus(row.principalDue);
    perCurrency.set(currency, entry);
  }

  const next = [...rows].sort((a, b) => a.dueDate.localeCompare(b.dueDate) || a.seq - b.seq)[0];
  const nextOwner = next ? byId.get(next.loanId)! : null;
  return {
    items: [...perCurrency.entries()]
      .sort(([a], [b]) => a.localeCompare(b))
      .map(([currency, entry]) => ({
        currency,
        borrowed: formatAmount(entry.borrowed, currency),
        lent: formatAmount(entry.lent, currency),
      })),
    next_due:
      next && nextOwner
        ? {
            loan: nextOwner.name,
            direction: nextOwner.direction,
            due_date: next.dueDate,
            total: formatAmount(new Decimal(next.principalDue).plus(next.interestDue), parseCurrency(nextOwner.currency)),
            currency: nextOwner.currency,
          }
        : null,
    overdue_count: rows.filter((row) => row.dueDate < asOf).length,
  };
}
```

```ts
import { accountById, resolveAccount, updateAccount } from '../ledger/accounts.js';
import { ensureCategory } from '../ledger/categories.js';
import { normalizeName } from '../ledger/normalize.js';
import { balancesFor, deleteTransaction, recordTransactions, recordTransfer } from '../ledger/transactions.js';
import { getUserSettings } from '../ledger/user-settings.js';

import { type Currency, parseCurrency } from '../money/currency.js';

import {
  type InstallmentRow,
  type LoanRow,
  installmentsOf,
  loadLoans,

  outstandingOf,
  parseRate,
  resolveLoan,

/** Danh mục hệ thống, tự tạo khi cần. */
export const LOAN_INTEREST_EXPENSE = 'Lãi & phí vay';
export const LOAN_INTEREST_INCOME = 'Lãi cho vay';

/** Khoá dòng khoản vay tới hết transaction: thao tác song song phải chờ và đọc lịch đã cập nhật. */
async function lockLoan(id: string): Promise<void> {
  await currentContext().tx.execute(sql`select 1 from loan where id = ${id} for update`);
}

async function activeLoan(ref: string): Promise<LoanRow> {
  const target = await resolveLoan(ref, { includeInactive: true });
  if (target.status !== 'active') {
    throw new DomainError(
      'LOAN_CLOSED',
      `Khoản vay "${target.name}" đã ${target.status === 'closed' ? 'tất toán' : 'bị huỷ'}.`,
    );
  }
  await lockLoan(target.id);
  return target;
}

/** Tính lại các kỳ chưa trả trên dư nợ mới; giữ số kỳ và ngày đến hạn. */
async function reschedule(target: LoanRow, due: readonly InstallmentRow[], outstanding: Decimal, annualRate: Decimal.Value) {
  const { tx } = currentContext();
  const amounts = computeInstallments({
    principal: outstanding,
    annualRate,
    count: due.length,
    method: target.method,
    currency: parseCurrency(target.currency),
  });
  for (const [index, row] of due.entries()) {
    await tx
      .update(loanInstallment)
      .set({ principalDue: amounts[index]!.principal.toFixed(), interestDue: amounts[index]!.interest.toFixed() })
      .where(eq(loanInstallment.id, row.id));
  }
}

export type LoanPaymentInput = {
  loan: string;
  paidOn?: string;
  /** Ví trả (vay) / ví nhận (cho vay); mặc định ví đã đặt cho khoản vay. */
  account?: string;
  /** Nhập đè gốc/lãi của kỳ (ví dụ lãi thực tế ngân hàng tính theo số ngày). */
  principal?: string;
  interest?: string;
  /** Trả thêm gốc trước hạn: các kỳ sau được tính lại. */
  extraPrincipal?: string;
  /** Phí (ví dụ phí trả trước hạn); chỉ cho khoản đi vay. */
  fee?: string;
};

export async function recordLoanPayment(input: LoanPaymentInput): Promise<{
  loan: LoanSummary;
  paid: InstallmentView;
  balances: AccountBalanceView[];
}> {
  const { tx } = currentContext();
  const target = await activeLoan(input.loan);
  const currency: Currency = parseCurrency(target.currency);
  const borrow = target.direction === 'borrow';
  const due = (await installmentsOf([target.id])).filter((row) => row.status === 'due');
  const next = due[0];
  if (!next) throw new DomainError('LOAN_CLOSED', `Khoản vay "${target.name}" không còn kỳ nào phải trả.`);

  const account = input.account
    ? await resolveAccount(input.account)
    : target.paymentAccountId
      ? await accountById(target.paymentAccountId)
      : null;
  if (!account) throw new DomainError('VALIDATION_ERROR', `Khoản vay "${target.name}" chưa có ví trả mặc định; truyền account.`);
  if (account.currency !== target.currency) {
    throw new DomainError('CURRENCY_MISMATCH', `Ví "${account.name}" dùng ${account.currency}, khoản vay dùng ${target.currency}.`);
  }
  if (input.fee !== undefined && !borrow) throw new DomainError('VALIDATION_ERROR', 'fee chỉ dùng cho khoản đi vay.');

  const principal = input.principal !== undefined ? parseAmount(input.principal, currency, { allowZero: true }) : new Decimal(next.principalDue);
  const interest = input.interest !== undefined ? parseAmount(input.interest, currency, { allowZero: true }) : new Decimal(next.interestDue);
  const extra = input.extraPrincipal !== undefined ? parseAmount(input.extraPrincipal, currency) : new Decimal(0);
  const fee = input.fee !== undefined ? parseAmount(input.fee, currency) : null;
  const principalTotal = principal.plus(extra);
  const outstanding = outstandingOf(due);
  if (principalTotal.greaterThan(outstanding)) {
    throw new DomainError(
      'OVERPAYMENT',
      `Gốc trả ${formatAmount(principalTotal, currency)} vượt dư nợ còn lại ${formatAmount(outstanding, currency)}.`,
      { outstanding_principal: formatAmount(outstanding, currency) },
    );
  }
  if (principalTotal.isZero() && interest.isZero()) throw new DomainError('VALIDATION_ERROR', 'Kỳ trả phải có gốc hoặc lãi.');

  const paidOn = input.paidOn ? parseIsoDate(input.paidOn, 'paid_on') : todayIn((await getUserSettings()).timezone);
  const label = `${target.name} kỳ ${next.seq}`;

  let principalTransferGroupId: string | null = null;
  if (!principalTotal.isZero()) {
    const amount = principalTotal.toFixed();
    const transfer = borrow
      ? await recordTransfer({ fromAccount: account.id, toAccount: target.accountId, amount, occurredOn: paidOn, note: `Trả gốc ${label}` })
      : await recordTransfer({ fromAccount: target.accountId, toAccount: account.id, amount, occurredOn: paidOn, note: `Thu gốc ${label}` });
    principalTransferGroupId = transfer.transfer_group_id;
  }

  const interestCategory = borrow
    ? await ensureCategory(LOAN_INTEREST_EXPENSE, 'expense')
    : await ensureCategory(LOAN_INTEREST_INCOME, 'income');
  let interestTransactionId: string | null = null;
  if (!interest.isZero()) {
    const { created } = await recordTransactions([
      {
        amount: interest.toFixed(),
        type: borrow ? 'expense' : 'income',
        account: account.id,
        category: interestCategory.id,
        occurredOn: paidOn,
        description: `Lãi ${label}`,
      },
    ]);
    interestTransactionId = created[0]!.id;
  }
  let feeTransactionId: string | null = null;
  if (fee) {
    const { created } = await recordTransactions([
      { amount: fee.toFixed(), type: 'expense', account: account.id, category: interestCategory.id, occurredOn: paidOn, description: `Phí ${label}` },
    ]);
    feeTransactionId = created[0]!.id;
  }

  const [paidRow] = await tx
    .update(loanInstallment)
    .set({
      status: 'paid',
      paidOn,
      principalPaid: principalTotal.toFixed(),
      interestPaid: interest.toFixed(),
      principalTransferGroupId,
      interestTransactionId,
      feeTransactionId,
    })
    .where(eq(loanInstallment.id, next.id))
    .returning();

  const rest = due.slice(1);
  const remaining = outstanding.minus(principalTotal);
  if (remaining.isZero()) {
    if (rest.length > 0) await tx.delete(loanInstallment).where(inArray(loanInstallment.id, rest.map((r) => r.id)));
    await tx.update(loan).set({ status: 'closed', closedAt: new Date() }).where(eq(loan.id, target.id));
  } else if (rest.length === 0) {
    // Kỳ cuối trả thiếu gốc: thêm một kỳ bù tháng sau.
    await tx.insert(loanInstallment).values({
      userId: target.userId,
      loanId: target.id,
      seq: next.seq + 1,
      dueDate: addOneMonth(next.dueDate),
      principalDue: remaining.toFixed(),
      interestDue: '0',
    });
  } else if (!principalTotal.equals(next.principalDue)) {
    await reschedule(target, rest, remaining, target.annualRate);
  }

  await recordAudit('loan.payment', 'loan_installment', [next.id]);
  const { schedule: _schedule, ...summary } = await loanDetail(target.id);
  return {
    loan: summary,
    paid: toInstallmentView(paidRow!, currency, paidOn),
    balances: await balancesFor([account.id, target.accountId]),
  };
}

export async function updateLoan(input: {
  loan: string;
  name?: string;
  annualRate?: string;
  paymentAccount?: string;
}): Promise<LoanDetail> {
  const { tx } = currentContext();
  const target = await activeLoan(input.loan);
  const changes: Partial<typeof loan.$inferInsert> = {};

  if (input.name !== undefined) {
    const name = requireName(input.name);
    const clash = (await loadLoans()).find((row) => row.id !== target.id && normalizeName(row.name) === normalizeName(name));
    if (clash) throw new DomainError('DUPLICATE_NAME', `Đã có khoản vay "${clash.name}".`);
    await updateAccount(target.accountId, { name });
    changes.name = name;
  }
  if (input.paymentAccount !== undefined) {
    const account = await resolveAccount(input.paymentAccount);
    if (account.currency !== target.currency) {
      throw new DomainError('CURRENCY_MISMATCH', `Ví "${account.name}" dùng ${account.currency}, khoản vay dùng ${target.currency}.`);
    }
    changes.paymentAccountId = account.id;
  }
  if (input.annualRate !== undefined) {
    const rate = parseRate(input.annualRate);
    changes.annualRate = rate.toFixed();
    const due = (await installmentsOf([target.id])).filter((row) => row.status === 'due');
    await reschedule(target, due, outstandingOf(due), rate);
  }
  if (Object.keys(changes).length === 0) throw new DomainError('VALIDATION_ERROR', 'Không có trường nào để sửa.');

  await tx.update(loan).set(changes).where(eq(loan.id, target.id));
  await recordAudit('loan.update', 'loan', [target.id]);
  return loanDetail(target.id);
}

export async function cancelLoan(ref: string): Promise<{ cancelled: true; loan: LoanSummary }> {
  const { tx } = currentContext();
  const target = await activeLoan(ref);
  const rows = await installmentsOf([target.id]);
  if (rows.some((row) => row.principalTransferGroupId || row.interestTransactionId || row.feeTransactionId)) {
    throw new DomainError(
      'LOAN_HAS_PAYMENTS',
      `Khoản vay "${target.name}" đã có kỳ trả được ghi nên không huỷ được. Nếu cần, xoá từng giao dịch bằng delete_transaction.`,
    );
  }
  if (target.disbursementTransactionId) await deleteTransaction(target.disbursementTransactionId);
  await updateAccount(target.accountId, { archived: true });
  await tx.update(loan).set({ status: 'cancelled', closedAt: new Date() }).where(eq(loan.id, target.id));
  await recordAudit('loan.cancel', 'loan', [target.id]);
  const { schedule: _schedule, ...summary } = await loanDetail(target.id);
  return { cancelled: true, loan: summary };
}
```

```ts
const loanRef = z.string().describe('Id hoặc tên khoản vay; không cần dấu');
const rateInput = z.string().describe('Lãi suất % năm dạng chuỗi, ví dụ "9.6"; "0" cho trả góp 0%');

const installmentView = z.object({
  seq: z.number(),
  due_date: z.string(),
  principal: z.string(),
  interest: z.string(),
  total: z.string(),
  status: z.enum(['due', 'paid', 'overdue']),
  paid_on: z.string().nullable(),
});

const ref = z.object({ id: z.string(), name: z.string() });

const loanSummary = z.object({
  id: z.string(),
  name: z.string(),
  direction: z.enum(LOAN_DIRECTIONS),
  method: z.enum(LOAN_METHODS),
  status: z.enum(LOAN_STATUSES),
  currency: z.string(),
  principal: z.string(),
  annual_rate: z.string(),
  term_months: z.number(),
  start_date: z.string(),
  first_due_date: z.string(),
  account: ref,
  payment_account: ref.nullable(),
  outstanding_principal: z.string(),
  paid_periods: z.number(),
  remaining_periods: z.number(),
  interest_paid: z.string(),
  interest_remaining: z.string(),
  next_due: installmentView.nullable(),
});

const loanDetail = { ...loanSummary.shape, schedule: z.array(installmentView) };

const nextDueText = (d: { next_due: { due_date: string; total: string } | null; currency: string }) =>
  d.next_due ? `kỳ tới ${d.next_due.due_date}: ${d.next_due.total} ${d.currency}` : 'không còn kỳ nào';

export function registerLoanTools(server: McpServer): void {
  server.registerTool(
    'create_loan',
    {
      title: 'Tạo khoản vay / trả góp / cho vay',
      description:
        'Tạo khoản vay (borrow) hoặc cho vay (lend), tự tạo ví nợ cùng tên và lịch trả hằng tháng. method: equal_principal = gốc đều, lãi trên dư nợ giảm dần (phổ biến ở ngân hàng); annuity = trả đều mỗi kỳ; flat = lãi phẳng trên gốc ban đầu (công ty tài chính). Trả góp 0% = flat với annual_rate "0". disbursement: cash (tiền vào/ra disbursement_account), purchase (mua hàng trả góp: ghi một khoản chi thuộc purchase_category), existing (khoản vay đã chạy: nhập paid_periods). Hỏi người dùng các thông số còn thiếu thay vì đoán.',
      inputSchema: {
        name: z.string().describe('Tên khoản vay, ví dụ "Vay mua xe", "Trả góp iPhone"'),
        direction: z.enum(LOAN_DIRECTIONS).describe('borrow = đi vay/trả góp, lend = cho người khác vay'),
        method: z.enum(LOAN_METHODS),
        principal: amountInput.describe('Số tiền gốc ban đầu'),
        annual_rate: rateInput,
        term_months: z.number().int().min(1).max(600).describe('Số kỳ (tháng)'),
        start_date: dateInput.optional().describe('Ngày giải ngân / mua; mặc định hôm nay'),
        first_due_date: dateInput.optional().describe('Ngày trả kỳ đầu; mặc định 1 tháng sau start_date'),
        disbursement: z.enum(['cash', 'purchase', 'existing']),
        disbursement_account: accountRef.optional().describe('cash: ví nhận tiền vay hoặc ví chi tiền cho vay'),
        purchase_category: categoryRef.optional().describe('purchase: danh mục chi của món hàng'),
        paid_periods: z.number().int().min(0).optional().describe('existing: số kỳ đã trả trước đây'),
        payment_account: accountRef.optional().describe('Ví trả nợ (hoặc nhận tiền trả) mặc định'),
        currency: z.string().optional(),
      },
      outputSchema: loanDetail,
      annotations: WRITE,
    },
    async (a) =>
      runTool(
        FINANCE_WRITE,
        () =>
          createLoan({
            name: a.name,
            direction: a.direction,
            method: a.method,
            principal: a.principal,
            annualRate: a.annual_rate,
            termMonths: a.term_months,
            startDate: a.start_date,
            firstDueDate: a.first_due_date,
            disbursement: a.disbursement,
            disbursementAccount: a.disbursement_account,
            purchaseCategory: a.purchase_category,
            paidPeriods: a.paid_periods,
            paymentAccount: a.payment_account,
            currency: a.currency,
          }),
        (d) => `Đã tạo "${d.name}": dư nợ ${d.outstanding_principal} ${d.currency}, ${d.remaining_periods} kỳ; ${nextDueText(d)}.`,
      ),
  );

  server.registerTool(
    'record_loan_payment',
    {
      title: 'Ghi một kỳ trả nợ',
      description:
        'Ghi kỳ kế tiếp của khoản vay: gốc thành chuyển khoản vào ví nợ, lãi thành khoản chi (hoặc khoản thu với cho vay). Mặc định theo lịch; truyền principal/interest khi số thật khác (ngân hàng tính lãi theo ngày). extra_principal = trả thêm gốc trước hạn, các kỳ sau được tính lại; hết dư nợ thì tất toán. Dùng tool này thay cho record_transfer khi trả nợ.',
      inputSchema: {
        loan: loanRef,
        paid_on: dateInput.optional(),
        account: accountRef.optional().describe('Ví trả (vay) hoặc ví nhận (cho vay); mặc định ví đã đặt'),
        principal: z.string().optional().describe('Gốc thực trả kỳ này (chuỗi số)'),
        interest: z.string().optional().describe('Lãi thực trả kỳ này (chuỗi số)'),
        extra_principal: amountInput.optional(),
        fee: amountInput.optional().describe('Phí (ví dụ phí trả trước hạn), chỉ cho khoản đi vay'),
      },
      outputSchema: { loan: loanSummary, paid: installmentView, balances: z.array(balanceView) },
      annotations: WRITE,
    },
    async (a) =>
      runTool(
        FINANCE_WRITE,
        () =>
          recordLoanPayment({
            loan: a.loan,
            paidOn: a.paid_on,
            account: a.account,
            principal: a.principal,
            interest: a.interest,
            extraPrincipal: a.extra_principal,
            fee: a.fee,
          }),
        (d) =>
          d.loan.status === 'closed'
            ? `Đã ghi kỳ ${d.paid.seq} và tất toán "${d.loan.name}".`
            : `Đã ghi kỳ ${d.paid.seq} (${d.paid.total} ${d.loan.currency}); còn nợ ${d.loan.outstanding_principal}, ${nextDueText(d.loan)}.`,
      ),
  );

  server.registerTool(
    'update_loan',
    {
      title: 'Sửa khoản vay',
      description: 'Đổi tên, ví trả mặc định, hoặc lãi suất mới (lãi thả nổi) — lãi suất mới tính lại các kỳ chưa trả trên dư nợ hiện tại.',
      inputSchema: {
        loan: loanRef,
        name: z.string().optional(),
        annual_rate: rateInput.optional(),
        payment_account: accountRef.optional(),
      },
      outputSchema: loanDetail,
      annotations: WRITE,
    },
    async (a) =>
      runTool(
        FINANCE_WRITE,
        () => updateLoan({ loan: a.loan, name: a.name, annualRate: a.annual_rate, paymentAccount: a.payment_account }),
        (d) => `Đã cập nhật "${d.name}" (lãi ${d.annual_rate}%/năm); ${nextDueText(d)}.`,
      ),
  );

  server.registerTool(
    'cancel_loan',
    {
      title: 'Huỷ khoản vay tạo nhầm',
      description: 'Huỷ khoản vay chưa có kỳ trả nào được ghi: xoá mềm giao dịch giải ngân, lưu trữ ví nợ. Hỏi xác nhận trước khi gọi.',
      inputSchema: { loan: loanRef },
      outputSchema: { cancelled: z.boolean(), loan: loanSummary },
      annotations: { readOnlyHint: false, destructiveHint: true, openWorldHint: false },
    },
    async ({ loan }) => runTool(FINANCE_WRITE, () => cancelLoan(loan), (d) => `Đã huỷ khoản vay "${d.loan.name}".`),
  );

  server.registerTool(
    'get_loan_status',
    {
      title: 'Tình hình khoản vay',
      description:
        'Không truyền loan: tóm tắt mọi khoản vay đang chạy (dư nợ, số kỳ đã/còn, lãi đã/còn phải trả, kỳ tới). Truyền loan: chi tiết kèm toàn bộ lịch trả. include_inactive để xem cả khoản đã tất toán/huỷ.',
      inputSchema: { loan: loanRef.optional(), include_inactive: z.boolean().optional() },
      outputSchema: { loans: z.array(loanSummary), schedule: z.array(installmentView).nullable() },
      annotations: READ_ONLY,
    },
    async ({ loan, include_inactive }) =>
      runTool(FINANCE_READ, () => getLoanStatus({ loan, includeInactive: include_inactive }), (d) =>
        d.loans.length === 0
          ? 'Không có khoản vay nào.'
          : d.loans.map((l) => `${l.name}: còn ${l.outstanding_principal} ${l.currency}, ${l.remaining_periods} kỳ`).join('; '),
      ),
  );

  server.registerTool(
    'list_upcoming_payments',
    {
      title: 'Kỳ trả sắp đến hạn',
      description:
        'Các kỳ quá hạn và đến hạn trong `days` ngày tới (mặc định 30) trên mọi khoản vay đang chạy, kèm tổng theo tiền tệ và chiều (borrow = phải trả, lend = sẽ được nhận). Dùng cho "tháng này phải trả những gì".',
      inputSchema: {
        as_of: dateInput.optional(),
        days: z.number().int().min(1).max(366).optional(),
      },
      outputSchema: {
        as_of: z.string(),
        until: z.string(),
        items: z.array(
          installmentView.extend({ loan: ref, direction: z.enum(LOAN_DIRECTIONS), currency: z.string() }),
        ),
        totals: z.array(
          z.object({
            currency: z.string(),
            direction: z.enum(LOAN_DIRECTIONS),
            principal: z.string(),
            interest: z.string(),
            total: z.string(),
          }),
        ),
        overdue_count: z.number(),
      },
      annotations: READ_ONLY,
    },
    async ({ as_of, days }) =>
      runTool(FINANCE_READ, () => listUpcomingPayments({ asOf: as_of, days }), (d) =>
        d.items.length === 0
          ? `Không có kỳ nào tới ${d.until}.`
          : `${d.items.length} kỳ tới ${d.until}${d.overdue_count ? `, ${d.overdue_count} kỳ quá hạn` : ''}.`,
      ),
  );
}
```

```ts
import type { Client } from '@modelcontextprotocol/sdk/client/index.js';
import type { NestExpressApplication } from '@nestjs/platform-express';
import { auth } from '../src/auth/auth.js';
import { startApp } from './app.js';
import { ALL_TOOLS, callError, callOk, connectMcp } from './mcp-client.js';
import { authorizeAndExchange, registerClient } from './oauth-flow.js';

const credentials = { email: 'loans-e2e@example.com', password: 'password-123' };

describe('loans over MCP', () => {
  let app: NestExpressApplication;
  let baseUrl: string;
  let client: Client;

  const balance = async (name: string): Promise<string> =>
    (await callOk(client, 'get_overview')).accounts.find((a: { name: string }) => a.name === name).balance;

  beforeAll(async () => {
    ({ app, baseUrl } = await startApp());
    await auth.api.signUpEmail({ body: { name: 'Loans E2E', ...credentials } });
    const clientId = await registerClient(baseUrl);
    const tokens = await authorizeAndExchange(baseUrl, clientId, credentials);
    client = await connectMcp(baseUrl, tokens.access_token);
    const { tools } = await client.listTools();
    expect(tools.map((t) => t.name).sort()).toEqual([...ALL_TOOLS].sort());
    await callOk(client, 'create_account', { type: 'bank', name: 'VCB', opening_balance: '10000000' });
  });

  afterAll(async () => {
    await client?.close();
    await app.close();
  });

  it('creates a car loan, pays an installment, changes the rate and lists what is due', async () => {
    const created = await callOk(client, 'create_loan', {
      name: 'Vay mua xe',
      direction: 'borrow',
      method: 'equal_principal',
      principal: '30000000',
      annual_rate: '9.6',
      term_months: 6,
      start_date: '2026-07-15',
      disbursement: 'cash',
      disbursement_account: 'VCB',
    });
    expect(created).toMatchObject({
      outstanding_principal: '30000000',
      account: expect.objectContaining({ name: 'Vay mua xe' }),
      next_due: expect.objectContaining({ seq: 1, due_date: '2026-08-15', total: '5240000' }),
    });
    expect(created.schedule).toHaveLength(6);

    const overview = await callOk(client, 'get_overview');
    expect(overview.accounts.find((a: { name: string }) => a.name === 'VCB').balance).toBe('40000000');
    expect(overview.debts.items).toEqual([{ currency: 'VND', borrowed: '30000000', lent: '0' }]);

    const paid = await callOk(client, 'record_loan_payment', { loan: 'vay mua xe', paid_on: '2026-08-15' });
    expect(paid.paid).toMatchObject({ seq: 1, status: 'paid', total: '5240000' });
    expect(paid.loan.outstanding_principal).toBe('25000000');
    expect(await balance('VCB')).toBe('34760000');

    // 25.000.000 còn 5 kỳ @ 12%: gốc 5.000.000, lãi kỳ 2 = 25.000.000 × 1% = 250.000.
    const updated = await callOk(client, 'update_loan', { loan: 'Vay mua xe', annual_rate: '12' });
    expect(updated.next_due).toMatchObject({ seq: 2, principal: '5000000', interest: '250000', total: '5250000' });

    const upcoming = await callOk(client, 'list_upcoming_payments', { as_of: '2026-09-20', days: 30 });
    expect(upcoming.items.map((i: any) => [i.seq, i.due_date, i.status, i.total])).toEqual([
      [2, '2026-09-15', 'overdue', '5250000'],
      [3, '2026-10-15', 'due', '5200000'],
    ]);
    expect(upcoming.totals).toEqual([
      { currency: 'VND', direction: 'borrow', principal: '10000000', interest: '450000', total: '10450000' },
    ]);

    const status = await callOk(client, 'get_loan_status', { loan: 'Vay mua xe' });
    expect(status.loans[0]).toMatchObject({ paid_periods: 1, remaining_periods: 5, annual_rate: '12' });
    expect(status.schedule[0].status).toBe('paid');
  });

  it('cancels a mistaken loan and protects loans with payments', async () => {
    await callOk(client, 'create_loan', {
      name: 'Vay nhầm',
      direction: 'borrow',
      method: 'flat',
      principal: '1000000',
      annual_rate: '0',
      term_months: 2,
      disbursement: 'cash',
      disbursement_account: 'VCB',
    });
    expect(await balance('VCB')).toBe('35760000');
    await callOk(client, 'cancel_loan', { loan: 'Vay nhầm' });
    expect(await balance('VCB')).toBe('34760000');
    expect((await callOk(client, 'get_loan_status')).loans.map((l: any) => l.name)).toEqual(['Vay mua xe']);

    expect((await callError(client, 'cancel_loan', { loan: 'Vay mua xe' })).code).toBe('LOAN_HAS_PAYMENTS');
    const overpay = await callError(client, 'record_loan_payment', { loan: 'Vay mua xe', extra_principal: '999999999' });
    expect(overpay.code).toBe('OVERPAYMENT');
    expect(overpay.details.outstanding_principal).toBe('25000000');
  });

  it('a read-only connection cannot record payments', async () => {
    const clientId = await registerClient(baseUrl);
    const tokens = await authorizeAndExchange(baseUrl, clientId, credentials, 'offline_access finance:read');
    const readOnly = await connectMcp(baseUrl, tokens.access_token);
    try {
      await readOnly.listTools();
      expect((await callError(readOnly, 'record_loan_payment', { loan: 'Vay mua xe' })).code).toBe('INSUFFICIENT_SCOPE');
      expect((await callOk(readOnly, 'get_loan_status')).loans[0].paid_periods).toBe(1);
    } finally {
      await readOnly.close();
    }
  });
});
```
